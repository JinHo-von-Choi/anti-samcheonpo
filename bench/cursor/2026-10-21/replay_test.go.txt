package live

import (
	"bytes"
	"encoding/json"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"github.com/JinHo-von-Choi/anti-samcheonpo/internal/adapter"
	"github.com/JinHo-von-Choi/anti-samcheonpo/internal/cost"
	"github.com/JinHo-von-Choi/anti-samcheonpo/internal/hookclient"
	"github.com/JinHo-von-Choi/anti-samcheonpo/internal/ledger"
)

// This diagnostic asserts the observed baseline, including its defects.
func TestCursorSupportProbe(t *testing.T) {
	dir := filepath.Join("..", "..", "bench", "cursor", "2026-10-21")
	read := func(name string) json.RawMessage {
		t.Helper()
		b, err := os.ReadFile(filepath.Join(dir, name))
		if err != nil {
			t.Fatal(err)
		}
		return b
	}
	before := read("shell-004-beforeShellExecution.json")
	after := read("shell-005-afterShellExecution.json")
	t.Run("captured_events", func(t *testing.T) {
		files, err := filepath.Glob(filepath.Join(dir, "*-*.json"))
		if err != nil {
			t.Fatal(err)
		}
		for _, file := range files {
			raw := read(filepath.Base(file))
			var m map[string]json.RawMessage
			if err := json.Unmarshal(raw, &m); err != nil {
				t.Fatal(err)
			}
			var event string
			if err := json.Unmarshal(m["hook_event_name"], &event); err != nil {
				t.Fatal(err)
			}
			events, inputs, err := translateIn("cursor", event, raw)
			if err != nil {
				t.Fatal(err)
			}
			idPreserved := false
			var nativeID string
			_ = json.Unmarshal(m["tool_use_id"], &nativeID)
			if len(inputs) > 0 && nativeID != "" {
				idPreserved = inputs[0].ToolUseID == nativeID
			}
			t.Logf("%s -> %v; native_id_preserved=%v", filepath.Base(file), events, idPreserved)
			if (event == "preToolUse" || event == "postToolUse") && len(events) != 0 {
				t.Fatal("baseline changed: generic event is handled")
			}
		}
	})
	t.Run("distinct_calls_collide", func(t *testing.T) {
		_, a, err := translateIn("cursor", "beforeShellExecution", before)
		if err != nil || len(a) != 1 {
			t.Fatalf("%v %v", a, err)
		}
		var m map[string]any
		_ = json.Unmarshal(before, &m)
		m["tool_use_id"] = "different-real-call"
		changed, _ := json.Marshal(m)
		_, b, _ := translateIn("cursor", "beforeShellExecution", changed)
		if a[0].ToolUseID != b[0].ToolUseID {
			t.Fatal("baseline changed: call identity no longer collides")
		}
		t.Log("synthetic second call: different native IDs produce the same internal ID")
	})
	t.Run("unknown_exit_in_ledger", func(t *testing.T) {
		root, home := t.TempDir(), t.TempDir()
		t.Setenv("SAMCHEONPO_HOME", home)
		if err := os.WriteFile(filepath.Join(root, ".samcheonpo.yml"), []byte("notify: {desktop: false}\niron_laws: {enabled: false}\n"), 0600); err != nil {
			t.Fatal(err)
		}
		db, err := ledger.Open(filepath.Join(home, "ledger.db"))
		if err != nil {
			t.Fatal(err)
		}
		defer db.Close()
		prices, err := cost.Load("")
		if err != nil {
			t.Fatal(err)
		}
		s, err := newSession("cursor-probe", "cursor", root, "", db, prices, adapter.Observe("cursor"))
		if err != nil {
			t.Fatal(err)
		}
		defer s.Finalize()
		d := &Daemon{sessions: map[string]*Session{s.ID: s}, db: db, prices: prices, versions: map[string]string{"cursor": "2026.10.19-6bbf2c4"}}
		for i, raw := range []json.RawMessage{before, after} {
			event := []string{"beforeShellExecution", "afterShellExecution"}[i]
			events, inputs, err := translateIn("cursor", event, raw)
			if err != nil || len(events) != 1 {
				t.Fatalf("%v %v", events, err)
			}
			in := inputs[0]
			in.SessionID = s.ID
			in.Cwd = s.Root
			in.TranscriptPath = ""
			if _, err := d.handleHook("cursor", events[0], in); err != nil {
				t.Fatal(err)
			}
		}
		if _, err := s.Finalize(); err != nil {
			t.Fatal(err)
		}
		var count, code int
		if err := d.db.QueryRow("SELECT COUNT(*),COALESCE(MIN(exit_code),-99) FROM event WHERE session_id=? AND tool='shell'", s.ID).Scan(&count, &code); err != nil {
			t.Fatal(err)
		}
		if count != 1 || code != 0 {
			t.Fatalf("baseline changed: count=%d exit=%d", count, code)
		}
		t.Logf("captured afterShellExecution has no exit_code; stored shell_count=%d exit_code=%d", count, code)
	})
	t.Run("output_and_fallback", func(t *testing.T) {
		root := t.TempDir()
		t.Setenv("SAMCHEONPO_HOME", root)
		t.Setenv("XDG_RUNTIME_DIR", root)
		t.Setenv("SAMCHEONPO_NO_SPAWN", "1")
		for _, event := range []string{"beforeShellExecution", "preToolUse"} {
			var out bytes.Buffer
			hookclient.MainAgent(event, "cursor", strings.NewReader(`{"conversation_id":"probe"}`), &out)
			t.Logf("daemon unavailable %s output=%q", event, out.String())
			if event == "preToolUse" && out.Len() != 0 {
				t.Fatal("baseline changed: generic allow fallback exists")
			}
			if event == "beforeShellExecution" && !strings.Contains(out.String(), `"permission":"allow"`) {
				t.Fatal("specialized allow fallback missing")
			}
		}
		out := translateOut("cursor", "postToolUse", "PostToolUse", json.RawMessage(`{"hookSpecificOutput":{"additionalContext":"probe-context"}}`), true)
		if len(out) != 0 {
			t.Fatalf("baseline changed: post context translated: %s", out)
		}
		t.Log("postToolUse additionalContext is dropped")
	})
	t.Run("capabilities", func(t *testing.T) {
		caps, tested := adapter.Resolve("cursor", "2026.10.19-6bbf2c4")
		if tested || caps.BlockPre || caps.InjectPost || caps.BlockStop {
			t.Fatalf("baseline changed: %+v %v", caps, tested)
		}
		t.Logf("tested=%v block_pre=%v inject_post=%v block_stop=%v", tested, caps.BlockPre, caps.InjectPost, caps.BlockStop)
	})
}
